\section{习题}

\subsection{列代数式}

\begin{enumerate}
    \item  笔记本一本 $a$ 元，买了 $5$ 本，一共多少元？
    \item 小明今年 $x$ 岁，妈妈今年 $y$ 岁，妈妈比小明大多少岁？（$y > x$）
    \item 边长为 $a$ 的正方形的周长是多少？面积是多少？
    \item 小明有 $x$ 个苹果，小红是小明个数的 $3$ 倍，小红有多少个苹果？
    \item 小明有 $x$ 个苹果，小红比小明个数的 $3$ 倍多 $5$ 个，小红有多少个苹果？
\end{enumerate}


\subsection{解方程}
\begin{enumerate}
    \item $x + 2.5 = 5$
    \item $3x - 2 = 2.5$
    \item $\frac{1}{2}x + 2 = 5$
    \item $\frac{3}{5}x + \frac{3}{4} = 5$
  
\end{enumerate}
